"""Captain-pinned two-log fidelity, bounded docs/whitespace and exact owned seal."""
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import shutil
import stat
import sys
import tempfile
import zipfile

sys.dont_write_bytecode = True
TEMP = Path('C:/Users/User/AppData/Local/Temp')
spec = importlib.util.spec_from_file_location('_dyd415_seal_owner', TEMP / 'dyd415-installed-finalize-20261030.py')
owner = importlib.util.module_from_spec(spec)
spec.loader.exec_module(owner)
driver = owner.driver
ROOT, NEXT = owner.ROOT, owner.NEXT
PIN = TEMP / 'dyd415-installed-raw-receipt-disposition-20261030.md'
CHECKPOINT = 'd7b4a1035c52697258bd849fdfa2bdc55190707fea427481a28a61e850c6ed56'
EXPECTED = {
    'npm-eof-red.stdout': 'e7a5f22362f1b32e0f24ac958ccdc5dfe35cab3d3e6265218d3ac34b3cf66187',
    'staged-whitespace.stdout': '0e11d3545be13f15f6ccd42f6a00c9d57e98c70d450efee6cb2d3462d2ab11bd'}


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def represent():
    output = NEXT / 'receipt-representation'
    output.mkdir(exist_ok=False)
    old = NEXT / 'installed-red/npm-eof-red.stdout'
    diagnostic = TEMP / 'dyd415-installed-finalize-n97kjb7h/staged-whitespace.stdout'
    paths = {'npm-eof-red.stdout': old, 'staged-whitespace.stdout': diagnostic}
    data = {member: path.read_bytes() for member, path in paths.items()}
    assert {member: digest(raw) for member, raw in data.items()} == EXPECTED
    assert driver.sha(NEXT / 'installed-checkpoint.json') == CHECKPOINT
    command = json.loads((NEXT / 'installed-red/npm-eof-red.command.json').read_bytes())
    failed = json.loads(diagnostic.with_name('staged-whitespace.command.json').read_bytes())
    assert command['exit'] == 1 and failed['exit'] == 2
    archive = output / 'npm-eof-red.zip'
    with zipfile.ZipFile(archive, 'x', compression=zipfile.ZIP_DEFLATED) as bundle:
        for member, raw in data.items():
            info = zipfile.ZipInfo(member)
            info.create_system = 3
            info.external_attr = (stat.S_IFREG | 0o600) << 16
            info.compress_type = zipfile.ZIP_DEFLATED
            bundle.writestr(info, raw)
    rows = [dict(oldPath=str(paths[member]), member=member, bytes=len(raw), sha256=EXPECTED[member],
        producerCommand=str(NEXT / 'installed-red/npm-eof-red.command.json') if member == 'npm-eof-red.stdout'
            else str(diagnostic.with_name('staged-whitespace.command.json')),
        producerExit=1 if member == 'npm-eof-red.stdout' else 2) for member, raw in data.items()]
    driver.save(output / 'npm-eof-red.manifest.json', dict(schema=1, sourceHead=driver.BASE,
        checkpoint=dict(path=str(NEXT / 'installed-checkpoint.json'), sha256=CHECKPOINT),
        captainPin=dict(path=str(PIN), sha256=driver.sha(PIN)),
        archive=dict(path=archive.relative_to(ROOT).as_posix(), sha256=driver.sha(archive), bytes=archive.stat().st_size),
        entries=rows, fidelityReceipt='fidelity.json',
        mapping={old.relative_to(ROOT).as_posix(): dict(archive=archive.relative_to(ROOT).as_posix(), member='npm-eof-red.stdout')},
        restoration='Fresh explicitly owned scratch only. Require exact two regular members, reject unsafe/duplicate/case-colliding/link names, extract exclusively and compare raw lengths/SHA256 against these pinned values. Old command/source/checkpoint hashes are unchanged; resolve only this exact old standalone path through its mapped member. Never execute either member.',
        signature='code-writer/model: gpt-6.1-sol/high'))
    scratch = Path(tempfile.mkdtemp(prefix='dyd415-two-log-fidelity-', dir=TEMP))
    verified = []
    with zipfile.ZipFile(archive) as bundle:
        infos = bundle.infolist()
        assert len(infos) == 2 and {info.filename for info in infos} == set(EXPECTED)
        seen = set()
        for info in infos:
            assert info.filename not in seen and info.filename.casefold() not in {name.casefold() for name in seen}
            seen.add(info.filename)
            assert info.filename in EXPECTED and '/' not in info.filename and '\\' not in info.filename
            assert stat.S_IFMT(info.external_attr >> 16) == stat.S_IFREG
            target = scratch / info.filename
            assert target.resolve().is_relative_to(scratch.resolve())
            with target.open('xb') as stream:
                stream.write(bundle.read(info))
            raw = target.read_bytes()
            assert len(raw) == len(data[info.filename]) == info.file_size
            assert digest(raw) == EXPECTED[info.filename] and raw == data[info.filename]
            verified.append(dict(member=info.filename, restoredPath=str(target), bytes=len(raw), sha256=digest(raw)))
    driver.save(output / 'fidelity.json', dict(archiveSha256=driver.sha(archive), checkedMembers=verified,
        exactMembers=True, regularFilesOnly=True, exclusiveExtraction=True, nonExecuting=True,
        sourceHashesPinnedBeforeRead=EXPECTED, originalStandaloneStillPresent=True,
        signature='code-writer/model: gpt-6.1-sol/high'))
    assert driver.sha(old) == EXPECTED['npm-eof-red.stdout'] and driver.sha(diagnostic) == EXPECTED['staged-whitespace.stdout']
    old.unlink()
    driver.save(output / 'removal.json', dict(removedOnly=old.relative_to(ROOT).as_posix(),
        oldRawSha256=EXPECTED['npm-eof-red.stdout'], exactBytesRetainedIn=archive.relative_to(ROOT).as_posix(),
        member='npm-eof-red.stdout', noOtherReceiptRemoval=True, temporaryDiagnosticUnchanged=True))
    return old, archive


def seal(output):
    baseline = (TEMP / 'dyd415-installed-finalize-n97kjb7h/owned-staged.stdout').read_text().splitlines()
    assert set(driver.git('diff', '--cached', '--name-only', '--no-renames').splitlines()) == set(baseline)
    assert len(baseline) == 239 and all(driver.owned(path) for path in baseline)
    _, binary, compiled = driver.installed_context()
    old, archive = represent()
    checkpoint = json.loads((NEXT / 'installed-checkpoint.json').read_bytes())
    pointer = NEXT / 'verification.md'
    for name, expected in checkpoint['exactReceiptHashes'].items():
        if ROOT / name == old:
            with zipfile.ZipFile(archive) as bundle:
                assert digest(bundle.read('npm-eof-red.stdout')) == expected
        elif ROOT / name != pointer:
            assert driver.sha(ROOT / name) == expected, 'Historical receipt changed: ' + name
    durable = NEXT / 'installed-seal'
    durable.mkdir(exist_ok=False)
    shutil.copy2(Path(__file__), durable / 'executed-driver.py.txt')
    driver.save(durable / 'source-binding.json', dict(parent=driver.BASE, compiledInputs=compiled,
        checkpointSha256=CHECKPOINT, representationManifestSha256=driver.sha(archive.with_name('npm-eof-red.manifest.json')),
        existingRawReceiptsUnchangedExceptExactRepresentation=True, noHealthyRuntimeOrTestRerun=True,
        captainPinSha256=driver.sha(PIN)))
    env = dict(os.environ)
    driver.run(durable, 'docs', [binary, 'check'], ROOT, env)

    def git(name, *args):
        driver.run(output, name, ['git', '-c', 'core.longpaths=true', *args], ROOT, env)
        return (output / (name + '.stdout')).read_text().rstrip('\r\n')

    additions = driver.git('ls-files', '--others', '--exclude-standard').splitlines()
    assert additions and all(driver.owned(path) for path in additions)
    # Only preverified owned staged files, this exact pointer and newly declared receipt paths.
    present = sorted({name for name in baseline + additions if name not in
        [driver.OLD_README, old.relative_to(ROOT).as_posix()] and (ROOT / name).is_file()})
    git('stage-scoped-files', 'add', '--', *present)
    git('stage-only-represented-removal', 'add', '-u', '--', old.relative_to(ROOT).as_posix())
    expected_paths = (set(baseline) - {old.relative_to(ROOT).as_posix()}) | set(additions)
    staged = git('staged-paths', 'diff', '--cached', '--name-only', '--no-renames').splitlines()
    assert set(staged) == expected_paths
    for name in staged:
        if name.startswith(driver.NATIVE + 'implementation3/'):
            assert digest(driver.git_bytes('show', ':' + name)) == driver.sha(ROOT / name), 'Receipt normalization: ' + name
    for name, base in [('parent', driver.BASE), ('contract', '3527ef642d9c0e1a4298cd13f22a53cf8a6a1a15'),
            ('master', '1318e93eabd500bbd45f243e4c99c3a10c2b004a')]:
        driver.run(durable, 'staged-whitespace-' + name,
            ['git', '-c', 'core.longpaths=true', 'diff', '--cached', '--check', base], ROOT, env)
    # Stage only the six new raw gate outputs after checking their completed contents.
    gate_files = [path.relative_to(ROOT).as_posix() for path in durable.glob('staged-whitespace-*')]
    git('stage-new-gate-receipts', 'add', '--', *gate_files)
    git('complete-staged-whitespace', 'diff', '--cached', '--check')
    git('commit', 'commit', '-m', 'DYD-415 implement: bind installed statechart skill and npm lifetime')
    candidate = git('candidate', 'rev-parse', 'HEAD')
    for name, base in [('parent', driver.BASE), ('contract', '3527ef642d9c0e1a4298cd13f22a53cf8a6a1a15'),
            ('master', '1318e93eabd500bbd45f243e4c99c3a10c2b004a')]:
        git('final-whitespace-' + name, 'diff', '--check', base, candidate)
    assert not git('clean', 'status', '--porcelain=v1', '--untracked-files=all')
    assert git('parents', 'rev-list', '--parents', '-n', '1', 'HEAD') == candidate + ' ' + driver.BASE
    assert driver.sha(NEXT / 'installed-checkpoint.json') == CHECKPOINT
    driver.representation(ROOT, driver.git('ls-tree', '-r', '--name-only', driver.HISTORY_BASE, '--', driver.NATIVE).splitlines())
    driver.navigation_case(ROOT)
    driver.save(output / 'candidate.json', dict(candidate=candidate, parent=driver.BASE, clean=True,
        checkpointSha256=CHECKPOINT, archiveSha256=driver.sha(archive),
        manifestSha256=driver.sha(archive.with_name('npm-eof-red.manifest.json')),
        fidelitySha256=driver.sha(archive.with_name('fidelity.json')), docsExit=0,
        stagedAndFinalWhitespaceExits=[0, 0, 0], noHealthyRuntimeOrTestRerun=True,
        failedGateAndOriginalHashesPreserved=True, noWhitespaceRuleSuppression=True,
        signature='code-writer/model: gpt-6.1-sol/high'))
    print('Clean candidate:', candidate, flush=True)


owner.finish = seal
if __name__ == '__main__':
    owner.main()
